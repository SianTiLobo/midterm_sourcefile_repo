const students = [
  {name: "Student 1", score: 92 },
  {name: "Student 2", score: 88 },
  {name: "Student 3", score: 85 },
  {name: "Student 4", score: 78 },
  {name: "Student 5", score: 72 },
  {name: "Student 6", score: 65 }
];

let total = 0;
let highest = students[0].score;
let lowest = students[0].score;
let passed = 0;
let failed = 0;

console.log("== STUDENT GRADE ANALYZER == \n");
console.log("All Student Scores: ");

students.forEach(s => {
  let grade;
  if (s.score >= 90) grade = "Excellent";
  else if (s.score >= 80) grade = "Very Good";
  else if (s.score >= 75) grade = "Good";
  else grade = "Failed";
  let status = s.score >= 75? "Passed" : "Failed";
  if (status === "Passed") passed++;
  else failed++;
  total+= s.score;
  if (s.score > highest) highest = s.score;
  if (s.score < lowest) lowest = s.score;
  console.log(`${s.name} - Score: ${s.score}, Grade: ${grade}, Status: ${status}`);
});

const average = total / students.length;
console.log("\n== SUMMARY ==");
console.log(`Total Score: ${total}`);
console.log(`Average Score: ${average.toFixed(2)}`);
console.log(`Highest Score: ${highest}`);
console.log(`Lowest Score: ${lowest}`);
console.log(`Number of Students Passed: ${passed}`);
console.log(`Number of Students Failed: ${failed}`);
